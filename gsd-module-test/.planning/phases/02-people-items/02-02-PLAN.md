addItem: (label, priceCents) =>
  set((state) => ({
    items: [
      ...state.items,
      { id: crypto.randomUUID(), label, price: priceCents, assignedTo: [] },
    ],
  })),

  set((state) => ({
    items: state.items.filter((i) => i.id !== id),
  })),
